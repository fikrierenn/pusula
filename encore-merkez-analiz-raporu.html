10. Hazır Sorgu Kütüphanesi

Aşağıdaki sorgular EncoreMerkez veritabanında doğrudan çalıştırılabilir. MCP üzerinden sql_query aracıyla veya SSMS'ten kullanılabilir. Tüm sorgular SQL Server 2012 (compat level 110) uyumludur.

Kural Hatırlatması: IsDeleted=0, IsValid=1, DocumentsTypeId IN (1,2) ciro için, IsChangeAmount=0 ödeme için, tarih CONVERT 104 formatında.

10.1 — Günlük Ciro + Kampanya Kırılımı

-- Son 7 gün, mağaza bazlı günlük ciro + kampanya kırılımı (3AL2ÖDE dahil)
;WITH cte_3al2ode AS (
    SELECT
        spc.SalesId,
        CAST(SUM(sp.TotalPrice + ABS(spc.DistributedAmount)) AS decimal(18,2)) AS BrutCiro,
        CAST(SUM(ABS(spc.DistributedAmount)) AS decimal(18,2)) AS Indirim,
        CAST(SUM(sp.TotalPrice) AS decimal(18,2)) AS NetCiro,
        COUNT(*) AS UrunSayisi
    FROM dbo.SalesProductCampaigns spc
    JOIN dbo.SalesProducts sp ON spc.SalesId = sp.SalesId AND spc.ProductSequence = sp.Sequence
    WHERE spc.CampaignId IN (1, 12) AND sp.IsValid = 1
    GROUP BY spc.SalesId
)
SELECT
    CONVERT(varchar, s.Date, 104) AS Tarih,
    st.Name AS Magaza,
    COUNT(DISTINCT s.Id) AS FisSayisi,
    CAST(SUM(s.GrossTotal) AS decimal(18,2)) AS BrutCiro,
    CAST(SUM(s.DiscountTotal) AS decimal(18,2)) AS Indirim,
    CAST(SUM(s.TotalAmount) AS decimal(18,2)) AS NetCiro,
    COUNT(DISTINCT CASE WHEN spc_all.SalesId IS NOT NULL THEN s.Id END) AS KampanyaliFis,
    COUNT(DISTINCT CASE WHEN spc_3al.SalesId IS NOT NULL THEN s.Id END) AS [3Al2Ode_Fis],
    ISNULL(SUM(spc_3al.UrunSayisi), 0) AS [3Al2Ode_Urun],
    CAST(ISNULL(SUM(spc_3al.BrutCiro), 0) AS decimal(18,2)) AS [3Al2Ode_BrutCiro],
    CAST(ISNULL(SUM(spc_3al.Indirim), 0) AS decimal(18,2)) AS [3Al2Ode_Indirim],
    CAST(ISNULL(SUM(spc_3al.NetCiro), 0) AS decimal(18,2)) AS [3Al2Ode_NetCiro]
FROM dbo.Sales s
JOIN dbo.Stores st ON s.StoresId = st.Id
LEFT JOIN (SELECT DISTINCT SalesId FROM dbo.SalesProductCampaigns) spc_all ON s.Id = spc_all.SalesId
LEFT JOIN cte_3al2ode spc_3al ON s.Id = spc_3al.SalesId
WHERE s.DocumentsTypeId IN (1,2)
  AND CAST(s.Date AS date) >= DATEADD(DAY, -7, CAST(GETDATE() AS date))
GROUP BY CONVERT(varchar, s.Date, 104), st.Name
ORDER BY Tarih DESC, Magaza

10.2 — Aylık Ciro Trend

-- Aylık brüt/net ciro, indirim oranı, fiş sayısı + 3AL2ÖDE kırılımı
;WITH cte_3al2ode AS (
    SELECT
        spc.SalesId,
        CAST(SUM(sp.TotalPrice + ABS(spc.DistributedAmount)) AS decimal(18,2)) AS BrutCiro,
        CAST(SUM(ABS(spc.DistributedAmount)) AS decimal(18,2)) AS Indirim,
        CAST(SUM(sp.TotalPrice) AS decimal(18,2)) AS NetCiro,
        COUNT(*) AS UrunSayisi
    FROM dbo.SalesProductCampaigns spc
    JOIN dbo.SalesProducts sp ON spc.SalesId = sp.SalesId AND spc.ProductSequence = sp.Sequence
    WHERE spc.CampaignId IN (1, 12) AND sp.IsValid = 1
    GROUP BY spc.SalesId
)
SELECT
    CONVERT(varchar(7), s.Date, 126) AS Ay,
    COUNT(*) AS FisSayisi,
    CAST(SUM(s.GrossTotal) AS decimal(18,2)) AS BrutCiro,
    CAST(SUM(s.DiscountTotal) AS decimal(18,2)) AS ToplamIndirim,
    CAST(SUM(s.TotalAmount) AS decimal(18,2)) AS NetCiro,
    CAST(SUM(s.VatTotal) AS decimal(18,2)) AS ToplamKDV,
    CASE WHEN SUM(s.GrossTotal) > 0
         THEN CAST(SUM(s.DiscountTotal) * 100.0 / SUM(s.GrossTotal) AS decimal(5,1))
         ELSE 0 END AS IndirimOrani,
    COUNT(DISTINCT CASE WHEN spc_3al.SalesId IS NOT NULL THEN s.Id END) AS [3Al2Ode_Fis],
    ISNULL(SUM(spc_3al.UrunSayisi), 0) AS [3Al2Ode_Urun],
    CAST(ISNULL(SUM(spc_3al.BrutCiro), 0) AS decimal(18,2)) AS [3Al2Ode_BrutCiro],
    CAST(ISNULL(SUM(spc_3al.Indirim), 0) AS decimal(18,2)) AS [3Al2Ode_Indirim],
    CAST(ISNULL(SUM(spc_3al.NetCiro), 0) AS decimal(18,2)) AS [3Al2Ode_NetCiro]
FROM dbo.Sales s
LEFT JOIN cte_3al2ode spc_3al ON s.Id = spc_3al.SalesId
WHERE s.DocumentsTypeId IN (1,2)
GROUP BY CONVERT(varchar(7), s.Date, 126)
ORDER BY Ay

10.4 — Ödeme Tipi Dağılımı

-- Tüm dönem veya tarih aralığı ile ödeme tipi dağılımı
SELECT
    pt.Name AS OdemeTipi,
    COUNT(*) AS IslemSayisi,
    CAST(SUM(sp.Amount) AS decimal(18,2)) AS Tutar,
    CASE WHEN (SELECT SUM(sp2.Amount) FROM dbo.SalesPayments sp2
          JOIN dbo.Sales s2 ON sp2.SalesId = s2.Id
          WHERE s2.DocumentsTypeId IN (1,2) AND sp2.IsChangeAmount = 0) > 0
         THEN CAST(SUM(sp.Amount) * 100.0 /
              (SELECT SUM(sp2.Amount) FROM dbo.SalesPayments sp2
               JOIN dbo.Sales s2 ON sp2.SalesId = s2.Id
               WHERE s2.DocumentsTypeId IN (1,2) AND sp2.IsChangeAmount = 0) AS decimal(5,1))
         ELSE 0 END AS YuzdePay
FROM dbo.SalesPayments sp
JOIN dbo.PaymentTypes pt ON sp.PaymentTypesId = pt.Id
JOIN dbo.Sales s ON sp.SalesId = s.Id
WHERE s.DocumentsTypeId IN (1,2) AND sp.IsChangeAmount = 0
GROUP BY pt.Name
ORDER BY Tutar DESC

    CAST(SUM(sp.TotalPrice) AS decimal(18,2)) AS ToplamTutar,
    CAST(SUM(sp.DiscountTotalCampaign) AS decimal(18,2)) AS KampanyaIndirim,
    ISNULL(SUM(spc_3al.UrunSayisi), 0) AS [3Al2Ode_Urun],
    CAST(ISNULL(SUM(spc_3al.BrutCiro), 0) AS decimal(18,2)) AS [3Al2Ode_BrutCiro],
    CAST(ISNULL(SUM(spc_3al.Indirim), 0) AS decimal(18,2)) AS [3Al2Ode_Indirim],
    CAST(ISNULL(SUM(spc_3al.NetCiro), 0) AS decimal(18,2)) AS [3Al2Ode_NetCiro]
FROM dbo.SalesProducts sp
JOIN dbo.Products p ON sp.ProductsId = p.Id
JOIN dbo.Sales s ON sp.SalesId = s.Id
LEFT JOIN dbo.ProductCategory pc ON p.CategoryId = pc.Id
LEFT JOIN dbo.ProductBrand pb ON p.BrandId = pb.Id
LEFT JOIN cte_3al2ode spc_3al ON sp.SalesId = spc_3al.SalesId AND sp.Sequence = spc_3al.ProductSequence
WHERE sp.IsValid = 1
  AND CAST(s.Date AS date) >= DATEADD(DAY, -30, CAST(GETDATE() AS date))
GROUP BY p.Code, p.Name, pc.Name, pb.Name
ORDER BY ToplamAdet DESC

-- Aktif kampanya tanımları ve kuralları
SELECT
    c.Id,
    c.Name AS KampanyaAd,
    c.CampaignTypeId AS TipId,
    CASE c.CampaignTypeId
        WHEN 1 THEN 'BuyN PayN'
        WHEN 2 THEN 'Yuzdesel/SabitFiyat'
        WHEN 5 THEN 'Tutar'
        ELSE 'Diger'
    END AS KampanyaTipi,
    CONVERT(varchar, c.BeginDate, 104) AS Baslangic,
    CONVERT(varchar, c.EndDate, 104) AS Bitis,
    c.BuyNPayN_1ConditionNum AS AlimKosul,
    c.BuyNPayN_1PayNum AS OdemeKosul,
    c.ConditionAmount AS KosulTutar,
    c.ConditionQuantity AS KosulAdet,
    c.Sequence AS Oncelik,
    c.ValidForAllStores AS TumMagazalar,
    c.IsActive
FROM dbo.Campaign c
WHERE c.IsActive = 1
ORDER BY c.Sequence, c.Name
